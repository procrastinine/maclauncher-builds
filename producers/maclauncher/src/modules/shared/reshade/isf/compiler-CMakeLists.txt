# The original wrapper forks force a Universal architecture default. Set each
# actual target to the enclosing package's selected CPUs after configuration.
function(select_architectures directory)
  get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
  foreach(target IN LISTS targets)
    get_target_property(type "${target}" TYPE)
    if(NOT type STREQUAL "INTERFACE_LIBRARY" AND NOT type STREQUAL "UTILITY")
      set_property(TARGET "${target}" PROPERTY OSX_ARCHITECTURES "${ISF_COMPILER_ARCHITECTURES}")
    endif()
  endforeach()
  get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
  foreach(child IN LISTS children)
    select_architectures("${child}")
  endforeach()
endfunction()
select_architectures("${CMAKE_CURRENT_SOURCE_DIR}")
