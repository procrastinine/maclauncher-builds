# Keep upstream warning suppression from hiding deployment-availability errors.
function(check_availability directory)
 get_property(options DIRECTORY "${directory}" PROPERTY COMPILE_OPTIONS)
 list(REMOVE_ITEM options -w)
 set_property(DIRECTORY "${directory}" PROPERTY COMPILE_OPTIONS "${options}")
 get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
 foreach(child IN LISTS children)
  check_availability("${child}")
 endforeach()
endfunction()
check_availability("${CMAKE_CURRENT_SOURCE_DIR}")
